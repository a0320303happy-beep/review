-- =========================================================
-- 設計提案評分｜第 4 版資料庫更新（內頁版式每個版式最多 4 張圖）
-- 用法：Supabase → SQL Editor → New query → 整份貼上 → Run（執行一次即可）
-- 需先執行過 update_v2、update_v3。不會刪除任何資料，原本的版式圖片會成為「第 1 張」。
-- =========================================================

-- 1. 圖片位置：封面 A/B/C，版式A 第1–4張（LA, LA2, LA3, LA4），版式B 第1–4張（LB, LB2, LB3, LB4）
alter table public.proposal_images drop constraint if exists proposal_images_slot_check;
alter table public.proposal_images
  add constraint proposal_images_slot_check
  check (slot in ('A', 'B', 'C', 'LA', 'LA2', 'LA3', 'LA4', 'LB', 'LB2', 'LB3', 'LB4'));

-- 2. 先列出有哪些圖（不含圖片內容），網頁再一張一張載入，手機開起來比較快
create or replace function public.get_image_list()
returns table (slot text, updated_at timestamptz)
language sql stable security definer set search_path = public as $$
  select slot, updated_at from proposal_images;
$$;

create or replace function public.get_image(p_slot text)
returns text
language sql stable security definer set search_path = public as $$
  select data_url from proposal_images where slot = p_slot;
$$;

revoke all on function public.get_image_list()  from public;
revoke all on function public.get_image(text)   from public;
grant execute on function public.get_image_list() to anon, authenticated;
grant execute on function public.get_image(text)  to anon, authenticated;

notify pgrst, 'reload schema';
