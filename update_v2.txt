-- =========================================================
-- 設計提案評分｜第 2 版資料庫更新
-- 用法：Supabase → SQL Editor → New query → 整份貼上 → Run（執行一次即可）
-- 內容：新增 3 分制評分表（封面 A/B/C ＋ 內頁版式 A/B），開放 C案與版式圖片。
-- 舊版的評分資料（5 分制）會保留在原本的表，不會被刪除。
-- =========================================================

-- 1. 新的評分表
create table if not exists public.ratings (
  id          uuid primary key,
  edit_token  text not null,
  created_at  timestamptz not null default now(),
  updated_at  timestamptz not null default now(),
  title       text not null check (char_length(title) between 1 and 60),
  scores      jsonb not null check (char_length(scores::text) <= 2000)
);
alter table public.ratings enable row level security;
revoke all on public.ratings from anon, authenticated;

-- 2. 圖片欄位：A / B / C（封面）、LA / LB（內頁版式）
alter table public.proposal_images drop constraint if exists proposal_images_slot_check;
alter table public.proposal_images
  add constraint proposal_images_slot_check check (slot in ('A', 'B', 'C', 'LA', 'LB'));

-- 3. 檢查分數格式：每一項都要有，而且只能是 1、2、3
create or replace function public.valid_scores(s jsonb)
returns boolean language plpgsql immutable as $$
declare
  spec jsonb := '{
    "cover":  {"items": ["A","B","C"], "criteria": ["identity","representative","readability"]},
    "layout": {"items": ["A","B"],     "criteria": ["whitespace","rhythm","hierarchy"]}
  }';
  sec text; it text; cr text; v jsonb;
begin
  if s is null or jsonb_typeof(s) <> 'object' then return false; end if;
  for sec in select jsonb_object_keys(spec) loop
    for it in select jsonb_array_elements_text(spec -> sec -> 'items') loop
      for cr in select jsonb_array_elements_text(spec -> sec -> 'criteria') loop
        v := s #> array[sec, it, cr];
        if v is null or jsonb_typeof(v) <> 'number' or (v #>> '{}')::numeric not in (1, 2, 3) then
          return false;
        end if;
      end loop;
    end loop;
  end loop;
  return true;
end $$;
revoke all on function public.valid_scores(jsonb) from public, anon, authenticated;

-- 4. 來賓送出 / 更新評分
create or replace function public.submit_rating(p jsonb)
returns uuid language plpgsql security definer set search_path = public as $$
declare
  rid uuid := (p->>'id')::uuid;
  tok text := p->>'token';
  t   text := btrim(coalesce(p->>'title', ''));
  s   jsonb := p->'scores';
begin
  if rid is null or tok is null or char_length(tok) < 16
     or char_length(t) not between 1 and 60
     or not public.valid_scores(s) then
    raise exception 'bad request' using errcode = '22023';
  end if;
  update ratings set updated_at = now(), title = t, scores = s
   where id = rid and edit_token = tok;
  if not found then
    insert into ratings (id, edit_token, title, scores) values (rid, tok, t, s);
  end if;
  return rid;
end $$;

-- 5. 後台：列出 / 刪除（需要密碼）
create or replace function public.admin_list_ratings(pw text)
returns table (id uuid, created_at timestamptz, updated_at timestamptz, title text, scores jsonb)
language plpgsql security definer set search_path = public as $$
begin
  perform public.assert_admin(pw);
  return query
    select r.id, r.created_at, r.updated_at, r.title, r.scores
    from ratings r order by r.updated_at desc;
end $$;

create or replace function public.admin_delete_rating(pw text, p_id uuid)
returns void language plpgsql security definer set search_path = public as $$
begin
  perform public.assert_admin(pw);
  delete from ratings where id = p_id;
end $$;

-- 6. 開放給網頁呼叫
revoke all on function public.submit_rating(jsonb)             from public;
revoke all on function public.admin_list_ratings(text)         from public;
revoke all on function public.admin_delete_rating(text, uuid)  from public;
grant execute on function public.submit_rating(jsonb)            to anon, authenticated;
grant execute on function public.admin_list_ratings(text)        to anon, authenticated;
grant execute on function public.admin_delete_rating(text, uuid) to anon, authenticated;

notify pgrst, 'reload schema';
